## 2.3 引力场计算

每个建筑向周围地块辐射引力，随距离衰减。地块的总引力场是周围所有建筑贡献的叠加。


function calculateGravityFromBuilding(building, targetCol, targetRow) {
  const distance = Math.abs(building.col - targetCol) + Math.abs(building.row - targetRow);
  if (distance > 5) return null; // 超过5格无影响

  const result = {};
  for (const [dim, value] of Object.entries(building.gravity)) {
    // 衰减公式：基础值 / (1 + distance × 0.5)
    result[dim] = Math.floor(value / (1 + distance * 0.5));
  }
  return result;
}

function calculateTileGravityField(col, row, gameState) {
  const field = { food:0, knowledge:0, comfort:0, adventure:0, culture:0, nature:0 };

  // 建筑贡献
  for (const building of gameState.buildings) {
    const contribution = calculateGravityFromBuilding(building, col, row);
    if (contribution) {
      for (const dim of Object.keys(field)) {
        field[dim] += contribution[dim] || 0;
      }
    }
  }

  // 地形自然贡献
  const tile = gameState.map[col][row];
  if (tile.gravity_bonus) {
    for (const [dim, val] of Object.entries(tile.gravity_bonus)) {
      field[dim] += val;
    }
  }

  // 活跃组合加成
  for (const combo of gameState.activeCombos) {
    if (isInComboRadius(col, row, combo)) {
      for (const dim of Object.keys(field)) {
        field[dim] += combo.gravity_bonus?.[dim] || 0;
      }
    }
  }

  return field;
}


## 2.4 引力匹配算法

游客/居民选择去哪里时，用偏好向量与目标引力场做匹配度计算：


function calculateAttraction(preferenceVector, targetGravity) {
  let score = 0;
  const dims = ['food','knowledge','comfort','adventure','culture','nature'];
  for (const dim of dims) {
    score += (preferenceVector[dim] || 0) * (targetGravity[dim] || 0);
  }
  return Math.min(100, score / 100); // 归一化到 0-100
}


## 2.5 引力热力图可视化

玩家点击左侧引力按钮或按数字键 1-6 切换查看对应维度的热力图叠加层：


function renderGravityHeatmap(ctx, dimension, gameState) {
  const color = GRAVITY_DIMENSIONS[dimension].color;
  const rgb = hexToRgb(color);

  for (let col = 0; col < gameState.mapWidth; col++) {
    for (let row = 0; row < gameState.mapHeight; row++) {
      const tile = gameState.map[col][row];
      if (!tile.discovered) continue;

      const gravity = calculateTileGravityField(col, row, gameState);
      const value = gravity[dimension] || 0;
      const intensity = Math.min(1, value / 100);

      if (intensity > 0.05) {
        const screenPos = gridToScreen(col, row);
        ctx.fillStyle = `rgba(${rgb.r},${rgb.g},${rgb.b},${intensity * 0.5})`;
        ctx.fillRect(screenPos.x, screenPos.y, TILE_SIZE, TILE_SIZE);

        if (intensity > 0.3) {
          ctx.fillStyle = 'white';
          ctx.font = '10px monospace';
          ctx.fillText(value.toString(), screenPos.x + 2, screenPos.y + 12);
        }
      }
    }
  }
}

## 3.4 地图生成


function generateMap(seed, width, height) {
  const noise = new SimplexNoise(seed);
  const map = [];

  for (let col = 0; col < width; col++) {
    map[col] = [];
    for (let row = 0; row < height; row++) {
      const elevation = noise.noise2D(col * 0.1, row * 0.1);
      const moisture  = noise.noise2D(col * 0.08 + 100, row * 0.08 + 100);
      const mineral   = noise.noise2D(col * 0.15 + 200, row * 0.15 + 200);

      let terrain;
      if (elevation > 0.75)                           terrain = 'mountain';
      else if (elevation < 0.2 && moisture > 0.6)     terrain = 'water';
      else if (moisture > 0.7)                         terrain = 'forest';
      else if (moisture > 0.5 && elevation < 0.5)     terrain = 'fertile';
      else if (mineral > 0.8)                          terrain = 'rocky';
      else if (mineral > 0.85 && elevation > 0.4)     terrain = 'crystal';
      else                                             terrain = 'plains';

      const tileType = TILE_TYPES[terrain];
      map[col][row] = {
        col, row,
        terrain,
        discovered: false,
        building_id: null,
        road_level: 0,
        decoration_id: null,
        fertility: randomInRange(tileType.fertility?.[0] ?? 0, tileType.fertility?.[1] ?? 0),
        mineral_richness: randomInRange(tileType.mineral_richness?.[0] ?? 0, tileType.mineral_richness?.[1] ?? 0),
        beauty: 0
      };
    }
  }

  // 初始可见区域：中心 12×10
  const cx = Math.floor(width / 2), cy = Math.floor(height / 2);
  for (let dx = -6; dx < 6; dx++) {
    for (let dy = -5; dy < 5; dy++) {
      if (map[cx+dx]?.[cy+dy]) map[cx+dx][cy+dy].discovered = true;
    }
  }

  // 开局保证规则
  guaranteeStartArea(map, cx, cy, {
    min_plains: 40,
    min_fertile: 6,
    max_water: 8,
    require_forest: true,
    require_water_adjacent: true
  });

  // 散布特殊地形（不在初始区域内）
  placeSpecialTiles(map, 'hot_spring', { count: [2,4], min_dist: 8 });
  placeSpecialTiles(map, 'ruins',      { count: [3,6], min_dist: 6 });
  placeSpecialTiles(map, 'anomaly',    { count: [1,3], min_dist: 12 });

  // 放置初始建筑
  map[cx][cy].building_id = 'b_headquarters_0';     // 指挥中心
  map[cx-1][cy].building_id = 'b_tent_0';           // 初始帐篷
  map[cx+1][cy].building_id = 'b_warehouse_0';      // 初始仓库
  map[cx][cy-1].road_level = 1;                      // 初始道路
  map[cx][cy+1].road_level = 1;

  return { map, width, height, seed, startX: cx, startY: cy };
}


## 3.5 地块交互规则


function canPlaceBuilding(template, col, row, gameState) {
  const tile = gameState.map[col][row];
  const tileType = TILE_TYPES[tile.terrain];

  if (!tile.discovered) return { ok: false, reason: '需要先探索该区域' };
  if (tile.building_id)  return { ok: false, reason: '已有建筑' };
  if (!tileType.buildable) return { ok: false, reason: `不能建在${tileType.name}上` };

  // 多格建筑检查
  for (let dx = 0; dx < template.size.w; dx++) {
    for (let dy = 0; dy < template.size.h; dy++) {
      const t = gameState.map[col+dx]?.[row+dy];
      if (!t || !t.discovered || t.building_id || !TILE_TYPES[t.terrain].buildable) {
        return { ok: false, reason: '占用区域不可用' };
      }
    }
  }

  // 地形要求
  if (template.terrain_requirement?.required_terrain) {
    if (!template.terrain_requirement.required_terrain.includes(tile.terrain)) {
      return { ok: false, reason: `需要建在指定地形上` };
    }
  }
  if (template.terrain_requirement?.required_adjacent) {
    const adj = getAdjacentTiles(col, row, gameState.map);
    if (!adj.some(t => template.terrain_requirement.required_adjacent.includes(t.terrain))) {
      return { ok: false, reason: `需要与指定地形相邻` };
    }
  }

  // 资源检查
  for (const [res, amt] of Object.entries(template.build_cost)) {
    if (res === 'special') continue;
    if ((gameState.resources[res] || 0) < amt) {
      return { ok: false, reason: `${res}不足` };
    }
  }
  for (const spec of (template.build_cost.special || [])) {
    if ((gameState.resources[spec.item_id] || 0) < spec.amount) {
      return { ok: false, reason: `${spec.item_id}不足` };
    }
  }

  // 前置科技检查
  if (template.unlock?.tech_required && !gameState.researched.includes(template.unlock.tech_required)) {
    return { ok: false, reason: '需要先研究对应科技' };
  }

  // 前置建筑检查
  if (template.unlock?.building_required && !gameState.buildings.some(b => b.type_id === template.unlock.building_required)) {
    return { ok: false, reason: '需要先建造前置建筑' };
  }

  // 人口要求
  if (template.unlock?.population_min && Object.keys(gameState.residents).length < template.unlock.population_min) {
    return { ok: false, reason: `需要至少${template.unlock.population_min}名居民` };
  }

  // 唯一建筑检查
  if (template.unique && gameState.buildings.some(b => b.type_id === template.id)) {
    return { ok: false, reason: '该建筑只能建造一个' };
  }

  return { ok: true };
}

## 5.3 生产流程与品质计算


function processProduction(building, gameState) {
  const recipe = building.production.current_recipe;
  if (!recipe) return;

  // 1. 检查输入材料
  for (const input of recipe.inputs) {
    const available = building.production.input_buffer[input.item_id] || 0;
    if (available < input.amount) {
      // 尝试从全局仓库拉取
      if ((gameState.resources[input.item_id] || 0) >= input.amount) {
        gameState.resources[input.item_id] -= input.amount;
        building.production.input_buffer[input.item_id] = (building.production.input_buffer[input.item_id] || 0) + input.amount;
      } else {
        return; // 材料不足，暂停
      }
    }
  }

  // 2. 计算生产速度
  const workers = building.assigned_workers.map(id => gameState.residents[id]);
  const avgWork = workers.reduce((s,w) => s + w.stats.work, 0) / Math.max(workers.length, 1);
  const speedMod =
    (1 + avgWork / 200) *
    (1 + (building.level - 1) * 0.2) *
    getTerrainBonus(building, gameState) *
    getComboSpeedBonus(building, gameState) *
    (gameState.season_effects?.farm_output || 1); // 季节加成（仅农业）

  // 3. 推进进度
  building.production.progress += (1 / recipe.base_time) * speedMod;

  // 4. 完成
  if (building.production.progress >= 1.0) {
    building.production.progress = 0;

    // 消耗材料
    for (const input of recipe.inputs) {
      building.production.input_buffer[input.item_id] -= input.amount;
    }

    // 计算品质
    const quality = calculateProductQuality(recipe, workers, building, gameState);

    // 生成产品实例
    const product = {
      id: generateUID(),
      item_id: recipe.output.item_id,
      name: recipe.name,
      quality_tier: quality.tier,
      quality_score: quality.score,
      price_modifier: quality.priceMod,
      gravity_tags: recipe.output_gravity,
      produced_by: building.id,
      produced_by_worker: workers[0]?.name || 'auto',
      turn_produced: gameState.currentDay,
      // AI异步填充
      ai_name: null,
      ai_description: null,
      ai_special_trait: null,
      ai_worker_comment: null
    };

    building.production.output_buffer.push(product);

    // 异步请求AI生成产品描述
    gameState.aiQueue.enqueue({
      request_type: 'product_description',
      priority: 4,
      context: { product, worker: workers[0], building, season: gameState.currentSeason },
      callback: (result) => {
        product.ai_name = result.name;
        product.ai_description = result.description;
        product.ai_special_trait = result.special_traits;
        product.ai_worker_comment = result.worker_comment;
      }
    });

    gameState.eventBus.emit('product_completed', { product, building });
  }
}

function calculateProductQuality(recipe, workers, building, gameState) {
  const f = recipe.quality_factors;
  const avgWork = workers.reduce((s,w) => s + w.stats.work, 0) / Math.max(workers.length, 1);

  const workerScore = Math.min(avgWork / 100, 1.0) * f.worker_stat_weight;
  const inputScore = 0.5 * f.input_quality_weight; // 简化：平均质量
  const facilityScore = (building.level / 5) * f.facility_level_weight;
  const comboScore = Math.min(getComboQualityBonus(building, gameState) / 100, 1.0) * f.combo_bonus_weight;

  const moodBonus = workers.reduce((s,w) => s + (w.mood - 50) / 500, 0) / Math.max(workers.length, 1);
  const random = 0.95 + Math.random() * 0.1;

  const total = Math.min(100, Math.max(0,
    (workerScore + inputScore + facilityScore + comboScore + moodBonus) * 100 * random
  ));

  let tier, priceMod;
  if      (total >= 95) { tier = 'S';  priceMod = 2.5; }
  else if (total >= 85) { tier = 'A+'; priceMod = 1.8; }
  else if (total >= 70) { tier = 'A';  priceMod = 1.4; }
  else if (total >= 50) { tier = 'B';  priceMod = 1.0; }
  else if (total >= 30) { tier = 'C';  priceMod = 0.7; }
  else                  { tier = 'D';  priceMod = 0.4; }

  return { score: total, tier, priceMod };
}

function updateResidentBehavior(resident, gameState) {
  const hour = gameState.currentHour;

  // 日程分配
  if (hour >= 6 && hour < 6 + resident.schedule.work_hours) {
    // 工作时间
    if (resident.state !== 'working' && resident.job.building_id) {
      moveTo(resident, resident.job.building_id, gameState);
    }
  } else if (hour >= 14 && hour < 14 + resident.schedule.free_hours) {
    // 自由时间：根据引力偏好选择去处
    if (resident.state === 'working' || resident.state === 'idle') {
      const dest = chooseLeisureDestination(resident, gameState);
      if (dest) moveTo(resident, dest, gameState);
    }
  } else {
    // 休息
    if (resident.state !== 'resting' && resident.home_building) {
      moveTo(resident, resident.home_building, gameState);
    }
  }
}

function chooseLeisureDestination(resident, gameState) {
  const candidates = gameState.buildings.filter(b =>
    ['commercial','decoration','tourism'].includes(BUILDING_TEMPLATES[b.type_id].category) &&
    b.is_active
  );

  let best = null, bestScore = -1;
  for (const b of candidates) {
    const gravity = calculateTileGravityField(b.col, b.row, gameState);
    let score = calculateAttraction(resident.gravity_affinity, gravity);
    const dist = manhattanDistance(resident.position, b);
    score *= (1 / (1 + dist * 0.15));
    if (score > bestScore) { bestScore = score; best = b; }
  }
  return best?.id || resident.home_building;
}

function gainExp(resident, amount) {
  resident.exp += amount;
  while (resident.exp >= resident.exp_to_next) {
    resident.exp -= resident.exp_to_next;
    resident.level++;
    resident.exp_to_next = Math.floor(resident.exp_to_next * 1.3);
    // 属性成长（根据当前工作类型）
    const growth = JOB_GROWTH[resident.job.role] || JOB_GROWTH.worker;
    for (const [stat, val] of Object.entries(growth)) {
      resident.stats[stat] += val + Math.floor(Math.random() * 3);
    }
  }
}

function checkTouristArrival(gameState) {
  if (!gameState.buildings.some(b => b.type_id === 'b_spaceport')) return; // 没宇宙港不来

  for (const species of Object.values(gameState.knownSpecies)) {
    const template = SPECIES[species.id];
    const daysSince = gameState.currentDay - (species.lastVisitDay || 0);
    const interval = Math.max(3, template.traits.visit_frequency * (1 - species.reputation.current / 200));

    if (daysSince >= interval) {
      const groupSize = randomInRange(...template.traits.group_size_range);
      const budget = template.traits.avg_budget + (Math.random() - 0.5) * 2 * template.traits.budget_variance;

      // 请求AI生成游客团
      gameState.aiQueue.enqueue({
        request_type: 'tourist_generation',
        priority: 2,
        context: {
          species: template,
          group_size: groupSize,
          total_budget: Math.round(budget),
          colony_gravity: calculateColonyOverallGravity(gameState),
          available_products: getAvailableProductTypes(gameState),
          season: gameState.currentSeason
        },
        callback: (result) => spawnTouristGroup(result, gameState)
      });

      species.lastVisitDay = gameState.currentDay;
    }
  }
}

function decideTouristRoute(tourist, gameState) {
  const destinations = gameState.buildings.filter(b => {
    const tmpl = BUILDING_TEMPLATES[b.type_id];
    return tmpl.commercial && b.is_active;
  });

  const scored = destinations.map(dest => {
    const gravity = calculateTileGravityField(dest.col, dest.row, gameState);
    let score = calculateAttraction(tourist.gravity_preference, gravity);
    const dist = manhattanDistance(gameState.spaceport_pos, dest);
    score *= (1 / (1 + dist * 0.1));
    return { building: dest, score };
  });

  scored.sort((a,b) => b.score - a.score);
  return scored.slice(0, Math.min(5, scored.length));
}

function touristConsume(tourist, building, product, gameState) {
  let price = product.base_price * product.price_modifier;
  price *= building.commercial?.base_price_modifier || 1;

  // 引力匹配溢价
  const attraction = calculateAttraction(tourist.gravity_preference, product.gravity_tags);
  price *= (1 + attraction / 200);

  // 组合加成
  price *= (1 + getComboBonus(building, 'tourist_spending', gameState) / 100);

  // 种族偏好
  if (product.preferred_by_species?.includes(tourist.species_id)) price *= 1.3;
  if (product.disliked_by_species?.includes(tourist.species_id)) price *= 0.5;

  const finalPrice = Math.round(price);
  if (finalPrice > tourist.remaining_budget * 1.2) return { purchased: false };

  tourist.remaining_budget -= finalPrice;
  gameState.resources.gold += finalPrice;

  return {
    purchased: true,
    price_paid: finalPrice,
    satisfaction: Math.round(attraction / 10),
    reputation_gain: Math.ceil(finalPrice / 100)
  };
}

function onBuildingPlaced(building, gameState) {
  const nearby = getBuildingsInRadius(building.col, building.row, 2, gameState);
  if (nearby.length < 1) return;

  const tagOverlap = countTagOverlaps(
    BUILDING_TEMPLATES[building.type_id].tags,
    nearby.map(b => BUILDING_TEMPLATES[b.type_id].tags)
  );

  if (tagOverlap >= 2) {
    gameState.aiQueue.enqueue({
      request_type: 'combo_check',
      priority: 2,
      context: {
        new_building: {
          id: building.id,
          type: building.type_id,
          category: BUILDING_TEMPLATES[building.type_id].category,
          tags: BUILDING_TEMPLATES[building.type_id].tags,
          level: building.level,
          gravity: BUILDING_TEMPLATES[building.type_id].gravity
        },
        adjacent_buildings: nearby.map(b => ({
          id: b.id,
          type: b.type_id,
          category: BUILDING_TEMPLATES[b.type_id].category,
          tags: BUILDING_TEMPLATES[b.type_id].tags,
          level: b.level,
          distance: manhattanDistance(building, b)
        })),
        existing_combos: gameState.activeCombos.map(c => c.id),
        colony_dominant_gravity: getDominantGravity(gameState),
        season: gameState.currentSeason
      },
      callback: (result) => {
        if (result.combo_detected) applyCombo(result.combo, gameState);
        if (result.near_miss_hint) showHint(result.near_miss_hint);
      }
    });
  }
}

## 10.2 战斗系统（极简回合制）


function executeBattle(party, enemies) {
  const allUnits = [
    ...party.map(r => ({ ...r, team:'player', alive:true })),
    ...enemies.map(e => ({ ...e, team:'enemy', alive:true }))
  ];
  allUnits.sort((a,b) => (b.stats?.speed||50) - (a.stats?.speed||50));

  const log = [];
  let round = 0;

  while (allUnits.some(u => u.team==='player' && u.alive) &&
         allUnits.some(u => u.team==='enemy' && u.alive) &&
         round < 50) {
    round++;
    for (const unit of allUnits) {
      if (!unit.alive) continue;
      const targets = allUnits.filter(u => u.team !== unit.team && u.alive);
      if (targets.length === 0) break;
      const target = targets.reduce((a,b) => a.stats.hp.current < b.stats.hp.current ? a : b);
      const damage = Math.max(1, unit.stats.combat - Math.floor((target.stats.combat||0)*0.3) + Math.floor(Math.random()*10)-5);
      target.stats.hp.current -= damage;
      if (target.stats.hp.current <= 0) { target.alive = false; target.stats.hp.current = 0; }
      log.push({ round, attacker:unit.name, target:target.name, damage, defeated:!target.alive });
    }
  }

  return {
    victory: allUnits.some(u => u.team==='player' && u.alive),
    rounds: round, log,
    survivors: allUnits.filter(u => u.team==='player').map(u => ({ id:u.id, hp:u.stats.hp.current })),
    loot: allUnits.some(u => u.team==='player' && u.alive) ? generateLoot(enemies) : []
  };
}

class AIRequestQueue {
  constructor(apiKey, maxConcurrent = 2) {
    this.queue = [];
    this.processing = 0;
    this.maxConcurrent = maxConcurrent;
    this.apiKey = apiKey;
    this.cache = new Map();
    this.fallbacks = {};
  }

  enqueue(request) {
    const cacheKey = this.getCacheKey(request);
    if (cacheKey && this.cache.has(cacheKey)) {
      request.callback(this.cache.get(cacheKey));
      return;
    }
    const idx = this.queue.findIndex(r => r.priority > request.priority);
    if (idx === -1) this.queue.push(request);
    else this.queue.splice(idx, 0, request);
    this.processNext();
  }

  async processNext() {
    if (this.processing >= this.maxConcurrent || this.queue.length === 0) return;
    this.processing++;
    const req = this.queue.shift();
    try {
      const systemPrompt = this.getSystemPrompt(req.request_type);
      const res = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: { 'Content-Type':'application/json', 'x-api-key':this.apiKey, 'anthropic-version':'2023-06-01' },
        body: JSON.stringify({
          model: 'claude-sonnet-4-20250514',
          max_tokens: req.max_tokens || 1000,
          system: systemPrompt,
          messages: [{ role:'user', content:JSON.stringify(req.context) }]
        })
      });
      const data = await res.json();
      const parsed = JSON.parse(data.content[0].text);
      if (this.validate(parsed, req.request_type)) {
        const cacheKey = this.getCacheKey(req);
        if (cacheKey) this.cache.set(cacheKey, parsed);
        req.callback(parsed);
      } else {
        req.callback(this.fallbacks[req.request_type] || {});
      }
    } catch (e) {
      console.warn('AI request failed:', e);
      req.callback(this.fallbacks[req.request_type] || {});
    } finally {
      this.processing--;
      this.processNext();
    }
  }

  getCacheKey(req) {
    if (req.request_type === 'combo_check') {
      const ids = [req.context.new_building.type,req.context.adjacent_buildings.map(b=>b.type)].sort().join('_');
      return `combo_${ids}`;
    }
    if (req.request_type === 'product_description') {
      return `product_${req.context.product.item_id}_${req.context.product.quality_tier}`;
    }
    return null;
  }
}

function startGameLoop(gameState) {
  let lastTime = Date.now();

  function loop() {
    const now = Date.now();
    const delta = now - lastTime;
    lastTime = now;

    if (gameState.speed === 0) { render(gameState); requestAnimationFrame(loop); return; }

    const ticks = Math.floor(delta / (TIME_CONFIG.ms_per_tick / gameState.speed));

    for (let i = 0; i < Math.min(ticks, 6); i++) { // 最多每帧处理6个tick防卡
      // 居民移动和行为
      for (const r of Object.values(gameState.residents)) {
        updateResidentMovement(r, gameState);
        updateResidentBehavior(r, gameState);
      }
      // 生产
      for (const b of gameState.buildings) {
        if (b.production?.current_recipe && b.is_active && b.is_connected) {
          processProduction(b, gameState);
        }
      }
      // 游客行为
      for (const t of gameState.activeTourists) {
        updateTouristBehavior(t, gameState);
      }
      // 时间推进
      gameState.currentHour++;
      if (gameState.currentHour >= 24) {
        gameState.currentHour = 0;
        gameState.currentDay++;
        dailyUpdate(gameState);
      }
    }

    render(gameState);
    requestAnimationFrame(loop);
  }
  requestAnimationFrame(loop);
}

function dailyUpdate(gameState) {
  // 1. 食物消耗
  const foodCost = Object.keys(gameState.residents).length * ECONOMY.daily_consumption.food_per_resident;
  gameState.resources.food = Math.max(0, gameState.resources.food - foodCost);

  // 2. 居民心情
  for (const r of Object.values(gameState.residents)) recalculateMood(r, gameState);

  // 3. 研究进度
  if (gameState.currentResearch) advanceResearch(gameState);

  // 4. 游客检查
  checkTouristArrival(gameState);

  // 5. 事件检查
  dailyEventCheck(gameState);

  // 6. 居民日记（30%概率）
  for (const r of Object.values(gameState.residents)) {
    if (Math.random() < 0.3) requestAIDiary(r, gameState);
  }

  // 7. 季节
  if (gameState.currentDay % TIME_CONFIG.days_per_season === 0) {
    const seasons = ['spring','summer','autumn','winter'];
    gameState.currentSeason = seasons[(seasons.indexOf(gameState.currentSeason)+1) % 4];
    applySeasonEffects(gameState);
  }

  // 8. 年终
  if (gameState.currentDay % (TIME_CONFIG.days_per_season * TIME_CONFIG.seasons_per_year) === 0) {
    gameState.currentYear++;
    requestAIAnnualReview(gameState);
  }

  // 9. 自动存档（每7天）
  if (gameState.currentDay % 7 === 0) saveGame(gameState);
}

function render(gameState) {
  const ctx = canvas.getContext('2d');
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // Layer 0: 地形
  renderTerrain(ctx, gameState);
  // Layer 1: 道路
  renderRoads(ctx, gameState);
  // Layer 2-3: 建筑（按Y排序实现遮挡）
  const sorted = [...gameState.buildings].sort((a,b) => a.row - b.row);
  for (const b of sorted) renderBuilding(ctx, b, gameState);
  // Layer 4: 居民和游客（按Y排序）
  const entities = [...Object.values(gameState.residents), ...gameState.activeTourists]
    .sort((a,b) => a.position.row - b.position.row);
  for (const e of entities) renderEntity(ctx, e, gameState);
  // Layer 5: 文字气泡和特效
  renderEffects(ctx, gameState);
  // Layer 6: 引力热力图（如果开启）
  if (gameState.activeHeatmap) renderGravityHeatmap(ctx, gameState.activeHeatmap, gameState);
  // Layer 7: 建筑放置预览
  if (gameState.placementMode) renderPlacementPreview(ctx, gameState);
}

# 开发建议：分阶段实现

| 阶段 | 模块 | 核心功能 | 预计工作量 |
|------|------|---------|-----------|
| **1** | 地图 + 建筑放置 | 网格渲染、地形生成、建筑放置/拆除、道路系统 | 先让东西能摆出来 |
| **2** | 居民 + 生产 | 居民寻路、生产链运行、品质计算、资源流转 | 让经济跑起来 |
| **3** | 引力系统 | 六维计算、热力图可视化、路径决策 | 让布局有意义 |
| **4** | 游客 | 种族定义、到访逻辑、消费计算、好感度 | 让金币流入 |
| **5** | AI接入 | 请求队列、组合判定、产品描述、日记 | 让世界活过来 |
| **6** | 探索 + 战斗 | 区域系统、编队、战斗、AI事件 | 让世界更大 |
| **7** | 科技 + 事件 | 科技树、事件系统、年终评比 | 让时间有意义 |
| **8** | 打磨 | 动画、音效、平衡性调整、存档优化 | 让体验完整 |
